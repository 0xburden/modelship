cmake_minimum_required(VERSION 3.24)
project(modelship_diffusion_probe LANGUAGES C CXX)

# No upstream CLI, HTTP server, Web UI, or dynamic plugin discovery.
if(NOT IS_ABSOLUTE "${SD_SOURCE}")
    message(FATAL_ERROR "Set SD_SOURCE to the verified upstream source directory")
endif()
if(SD_USE_SYSTEM_GGML OR SD_USE_UPSTREAM_GGML OR SD_BUILD_SHARED_GGML_LIB OR GGML_BACKEND_DL)
    message(FATAL_ERROR "The probe requires the patched, statically linked ggml source")
endif()
if(SD_GGML_SOURCE_DIR)
    set(MODELSHIP_GGML_SOURCE "${SD_GGML_SOURCE_DIR}")
else()
    set(MODELSHIP_GGML_SOURCE "${SD_SOURCE}/ggml")
endif()
if(NOT IS_ABSOLUTE "${MODELSHIP_GGML_SOURCE}")
    message(FATAL_ERROR "Use an absolute path to a disposable ggml source tree")
endif()
include(cuda-placement.cmake)
include(cuda-residency.cmake)
add_subdirectory("${SD_SOURCE}" sd EXCLUDE_FROM_ALL)

add_executable(modelship-placement-test EXCLUDE_FROM_ALL placement-test.cpp)
set_target_properties(modelship-placement-test PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
target_link_libraries(modelship-placement-test PRIVATE ggml)
target_include_directories(modelship-placement-test PRIVATE "${MODELSHIP_GGML_SOURCE}/src")
target_compile_features(modelship-placement-test PRIVATE cxx_std_17)
add_executable(modelship-sd-probe probe.cpp)
set_target_properties(modelship-sd-probe PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
target_link_libraries(modelship-sd-probe PRIVATE stable-diffusion)
target_compile_features(modelship-sd-probe PRIVATE cxx_std_17)
if(SD_CUDA)
    target_compile_definitions(modelship-sd-probe PRIVATE MODELSHIP_SD_CUDA)
endif()

add_executable(modelship-sd-protocol-test EXCLUDE_FROM_ALL worker-protocol-test.cpp)
set_target_properties(modelship-sd-protocol-test PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
target_compile_features(modelship-sd-protocol-test PRIVATE cxx_std_17)

add_executable(modelship-sd-worker worker.cpp)
set_target_properties(modelship-sd-worker PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
target_link_libraries(modelship-sd-worker PRIVATE stable-diffusion)
target_compile_features(modelship-sd-worker PRIVATE cxx_std_17)
if(SD_CUDA)
    find_package(CUDAToolkit REQUIRED)
    target_link_libraries(modelship-sd-worker PRIVATE CUDA::cudart)
    target_compile_definitions(modelship-sd-worker PRIVATE MODELSHIP_SD_CUDA)
endif()

if(SD_CUDA)
    # Separate executable and discriminator: not a replacement for the
    # already validated generation-only worker or its installed protocol.
    add_executable(modelship-sd-edit-candidate worker.cpp)
    set_target_properties(modelship-sd-edit-candidate PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
    target_link_libraries(modelship-sd-edit-candidate PRIVATE stable-diffusion CUDA::cudart)
    target_compile_features(modelship-sd-edit-candidate PRIVATE cxx_std_17)
    target_compile_definitions(modelship-sd-edit-candidate PRIVATE MODELSHIP_SD_CUDA MODELSHIP_SD_EDIT)

    add_executable(modelship-sd-edit-worker worker.cpp)
    set_target_properties(modelship-sd-edit-worker PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
    target_link_libraries(modelship-sd-edit-worker PRIVATE stable-diffusion CUDA::cudart)
    target_compile_features(modelship-sd-edit-worker PRIVATE cxx_std_17)
    target_compile_definitions(modelship-sd-edit-worker PRIVATE MODELSHIP_SD_CUDA MODELSHIP_SD_EDIT MODELSHIP_SD_EDIT_SERVICE)
endif()
